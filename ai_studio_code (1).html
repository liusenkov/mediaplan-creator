async function exportToGoogleSheets() { const btn = document.getElementById('exportBtn'); const originalText = btn.innerHTML; btn.disabled = true; btn.innerHTML = ' Создание...'; const cards = document.querySelectorAll('.position-card'); const data = { positions: [] }; cards.forEach(card => { data.positions.push({ targeting: `РФ, ${Array.from(card.querySelector('.geo-input').selectedOptions).map(o => o.text).join(', ')}; ${card.querySelector('.gender-input').value}, ${card.querySelector('.age-from').value}-${card.querySelector('.age-to').value}`, format: card.querySelector('.goal-input').value, period: card.querySelector('.period-val').value + ' ' + card.querySelector('.period-unit').options[card.querySelector('.period-unit').selectedIndex].text, budget: card.querySelector('.kpi-budget').value, cpm: card.querySelector('.kpi-cpm').value, imps: Math.round((card.querySelector('.kpi-budget').value / card.querySelector('.kpi-cpm').value) * 1000) }); }); try { // Используем встроенный в Google механизм перенаправления const response = await fetch(GOOGLE_SCRIPT_URL, { method: 'POST', body: JSON.stringify(data) }); const fileUrl = await response.text(); if (fileUrl.startsWith('https://')) { // Открываем созданный файл в новой вкладке window.open(fileUrl, '_blank'); } else { alert('Файл создан на вашем Google Диске'); } } catch (e) { // Из-за CORS браузер может блокировать чтение ответа, // но файл все равно создастся. alert('Запрос отправлен. Если вкладка не открылась, проверьте "Недавние" на Google Диске.'); } finally { btn.disabled = false; btn.innerHTML = originalText; lucide.createIcons(); } }
